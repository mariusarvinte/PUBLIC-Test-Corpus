add_custom_command(
	OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/clar.suite ${CMAKE_CURRENT_BINARY_DIR}/clar_suite.h
	COMMAND ${PYTHON_EXECUTABLE} ${CLAR_PATH}/generate.py -o "${CMAKE_CURRENT_BINARY_DIR}" -f -xonline -xstress -xperf .
	DEPENDS ${SRC_TEST}
	WORKING_DIRECTORY ${TEST_PATH}
)

set_source_files_properties(
	${CLAR_PATH}/clar.c
	PROPERTIES OBJECT_DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/clar.suite)

message(STATUS "LIBGIT2_LINK_PATH is set to: '${LIBGIT2_LINK_PATH}'")
add_executable(libgit2_tests ${SRC_CLAR} ${SRC_TEST})
set_target_properties(libgit2_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR})
target_include_directories(libgit2_tests PRIVATE ${TEST_INCLUDES} ${LIBGIT2_INCLUDES} ${LIBGIT2_DEPENDENCY_INCLUDES})
target_include_directories(libgit2_tests SYSTEM PRIVATE ${LIBGIT2_SYSTEM_INCLUDES})

# If the user specified a custom path to link against do that
# If not specified then default to the C generated libgit2 package
if(LIBGIT2_LINK_PATH)
	message(STATUS "Testing will link against custom one")
	target_link_libraries(libgit2_tests ${LIBGIT2_LINK_PATH} util ${LIBGIT2_SYSTEM_LIBS})
else()
	message(STATUS "Tests will link against built libgit2package library")
	target_link_libraries(libgit2_tests libgit2package util ${LIBGIT2_SYSTEM_LIBS})
	add_dependencies(libgit2_tests libgit2package)
endif()
